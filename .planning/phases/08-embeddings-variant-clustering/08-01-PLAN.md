**Key context from 08-RESEARCH.md:**
- Cohere Embed v4 accepts up to 96 items per call, 5 MB per image
- Matryoshka dimensions: default 1536, can request 256/512/1024 to cut bandwidth
- `inputType: "image"` with `images: ["data:<mime>;base64,<b64>"]` array
- Cohere returns normalized vectors → dot product equals cosine similarity (but implement norm division anyway for safety since dimension choice affects this)
- Union-find for n ≤ 200 is O(n²/2) = trivial

    ```typescript
    import { getCohereClient } from "./cohere-client";

    export interface EmbedImageInput {
      base64: string;
      mimeType: string;
    }

    export const COHERE_EMBED_MODEL = "embed-v4.0";
    export const COHERE_EMBED_BATCH_SIZE = 96;
    export const COHERE_EMBED_DIMENSION = 512; // Matryoshka — bandwidth-friendly; quality check in Plan 08-02

    export function cosineSimilarity(a: number[], b: number[]): number { ... }

    export function clusterByCosine(
      vectors: number[][],
      threshold: number,
    ): number[][] {
      // union-find implementation from 08-RESEARCH.md Code Examples section.
      // After bucketing, sort each bucket ascending by index, then sort buckets by their
      // smallest index ascending. Determinism is REQUIRED for test assertions and for
      // downstream label derivation in Plan 08-02.
    }

    export async function embedImagesCohere(
      images: EmbedImageInput[],
      opts: { dimension?: 256 | 512 | 1024 | 1536 } = {},
    ): Promise<number[][]> {
      const dimension = opts.dimension ?? COHERE_EMBED_DIMENSION;
      const client = getCohereClient();
      const results: number[][] = [];
      for (let i = 0; i < images.length; i += COHERE_EMBED_BATCH_SIZE) {
        const slice = images.slice(i, i + COHERE_EMBED_BATCH_SIZE);
        const res = await client.embed({
          model: COHERE_EMBED_MODEL,
          inputType: "image",
          embeddingTypes: ["float"],
          // outputDimension name — verify against installed cohere-ai SDK typings at execution time;
          // if the SDK uses a different field name (e.g. `embedDimensions`), adapt the call site
          // and leave a comment referencing the SDK version.
          outputDimension: dimension,
          images: slice.map((img) => `data:${img.mimeType};base64,${img.base64}`),
        });
        const vectors = res.embeddings?.float;
        if (!vectors || vectors.length !== slice.length) {
          throw new Error(
            `Cohere embed returned ${vectors?.length ?? 0} vectors for batch of ${slice.length}`,
          );
        }
        results.push(...vectors);
      }
      return results;
    }
    ```

    Implementation notes:
    - Use `Math.hypot` or a norm precompute to guard against non-normalized vectors even though Cohere float embeddings are typically L2-normalized.
    - Union-find: path compression in `find`, union-by-anything is fine for n ≤ 200.
    - Do NOT import `p-limit` here — batching is sequential per 08-CONTEXT.md decision.
    - Do NOT add retry logic in this file — Plan 08-02 owns the retry/fallback policy.
