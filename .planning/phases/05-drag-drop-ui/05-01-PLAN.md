```typescript
// FileItem — unchanged from upload-zone.tsx current definition
interface FileItem {
  id: string;        // crypto.randomUUID()
  file: File;        // actual File object (reconstructed from Blob on restore)
  url: string;       // URL.createObjectURL() — revoked on unmount
}

// Group — NEW type replacing FileItem[][] flat array
interface Group {
  id: string;          // stable UUID — used as dnd-kit droppable id, NOT positional index
  items: FileItem[];   // ordered; items[0] = hero image
  maxImages: number;   // per-group override (default = globalGroupSize at creation time)
}

// IDB storage types (internal to hook)
interface BlobRecord {
  id: string;        // matches FileItem.id
  blob: Blob;        // actual binary — stored via structured clone, never JSON
  filename: string;
  mimeType: string;
  savedAt: number;   // Date.now() — used for 24h expiry check
}

interface GroupRecord {
  id: "groups-v1";       // singleton key — single groups document
  groups: { id: string; itemIds: string[]; maxImages: number }[][];
  savedAt: number;
}

// Hook return type
interface StagedImagesHook {
  loadStaged: () => Promise<{ groups: Group[]; urlsCreated: string[] }>;
  saveBlob: (id: string, file: File) => Promise<void>;
  deleteBlob: (id: string) => Promise<void>;
  saveGroups: (groups: Group[]) => Promise<void>;
  clearAll: () => Promise<void>;
}
```

    Create `client/src/hooks/use-staged-images.ts` with the following implementation:

    1. Export `FileItem`, `Group`, `BlobRecord` types (these are the shared contracts for 05-02 and 05-03).

    2. DB schema: database name `"staged-images-db"`, version 1. Two stores:
       - `"blobs"` with `keyPath: "id"` — stores BlobRecord objects
       - `"groups"` with `keyPath: "id"` — stores the singleton GroupRecord with id `"groups-v1"`

    3. Open DB helper (module-level singleton using a promise cache — call `openDB` once, cache the promise):
       ```typescript
       let dbPromise: ReturnType<typeof openDB<StagedDB>> | null = null;
       function getDB() {
         if (!dbPromise) dbPromise = openDB<StagedDB>('staged-images-db', 1, {
           upgrade(db) {
             db.createObjectStore('blobs', { keyPath: 'id' });
             db.createObjectStore('groups', { keyPath: 'id' });
           },
         });
         return dbPromise;
       }
       ```
       Wrap all IDB calls in try/catch — on failure log `console.warn('[use-staged-images] IDB unavailable:', err)` and return a no-op result. This handles Safari private mode.

    4. `loadStaged()`:
       - Fetch all BlobRecord from `"blobs"` store
       - Filter out records where `Date.now() - savedAt >= 24 * 60 * 60 * 1000` (expired)
       - Delete expired records in a loop: `await db.delete('blobs', b.id)`
       - Fetch GroupRecord from `"groups"` store by key `"groups-v1"`
       - If no groups record OR no fresh blobs → return `{ groups: [], urlsCreated: [] }`
       - Build a `Map<string, BlobRecord>` from fresh blobs (keyed by id)
       - Reconstruct `FileItem[]` by creating `URL.createObjectURL(blobRecord.blob)` for each blob; collect all created URLs in `urlsCreated` array (caller adds these to `urlsRef` for cleanup)
       - Reconstruct `Group[]` from the groups record: for each stored group array, map itemIds to FileItems using the blob map; skip items whose id is not in blobMap (blob may have been deleted)
       - Filter out empty groups
       - Return `{ groups: reconstructedGroups, urlsCreated }`

    5. `saveBlob(id, file)`:
       - `await db.put('blobs', { id, blob: file, filename: file.name, mimeType: file.type, savedAt: Date.now() })`
       - NOTE: Pass the `File` directly as `blob` — `File` extends `Blob`, structured clone handles it

    6. `deleteBlob(id)`:
       - `await db.delete('blobs', id)`

    7. `saveGroups(groups)`:
       - Convert `Group[]` to the serializable shape: `{ id: string; itemIds: string[]; maxImages: number }[]`
       - Note: groups is a flat array (not nested) — each Group has items[] array
       - `await db.put('groups', { id: 'groups-v1', groups: serialized, savedAt: Date.now() })`

    8. `clearAll()`:
       - `await db.clear('blobs')`
       - `await db.delete('groups', 'groups-v1')`

    9. Export a `useStagedImages()` hook that returns all five functions as stable references (use `useCallback` or just return object literals — these are async functions, no render-loop risk).

    TypeScript: Use idb's `DBSchema` type parameter for full type safety on store names and value types.

    IMPORTANT — do NOT store File[] arrays in the groups record. Only store `itemIds: string[]`. The blob store is the source of truth for binary data.
