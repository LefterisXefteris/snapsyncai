**B. Update the unlock error handler to detect weekly cap errors:**

When the unlock endpoint returns 403 with `message: "Weekly limit reached"`, the error object or response will contain `weeklyLimit`, `used`, and `resetsAt`. Update the error handler to show a specific toast for this case:

```typescript
onError: (error: any) => {
  // Check for weekly cap 403
  if (error?.weeklyLimit !== undefined || error?.message === "Weekly limit reached") {
    const resetsAt = error?.resetsAt
      ? new Date(error.resetsAt).toLocaleDateString('en-GB', { weekday: 'long', month: 'short', day: 'numeric' })
      : 'next Monday';
    toast({
      title: "Weekly limit reached",
      description: `You've used ${error?.used ?? error?.weeklyLimit ?? 30} of ${error?.weeklyLimit ?? 30} products this week. Your limit resets ${resetsAt}.`,
      variant: "destructive",
    });
    return;
  }
  // Existing subscription-required 403 handling
  if (error?.message === "Subscription required") {
    toast({
      title: "Subscription required",
      description: "Subscribe to SnapSync AI to unlock AI product analysis.",
      variant: "destructive",
    });
    return;
  }
  // Default error
  toast({
    title: "Unlock failed",
    description: error?.message ?? "Something went wrong. Please try again.",
    variant: "destructive",
  });
},
```
